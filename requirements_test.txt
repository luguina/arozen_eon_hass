# Test dependencies, pinned to what the Home Assistant box actually runs. The point of
# the suite is to catch a change breaking against the core this integration is deployed
# on, not against whatever pip happened to resolve the day CI ran.
#
# These need Python 3.14. On 3.12 pip silently backtracks to Home Assistant 2025.1.4 and
# on 3.13 to 2026.2.3 - both import cleanly and then test a core nobody runs. The venv
# used for probing (.venv, tinytuya on 3.12) is NOT the test venv; use .venv-test, built
# from the uv-managed 3.14.
homeassistant==2026.8.1
pytest==9.1.1
pytest-asyncio==1.3.0

# The integration's own manifest requirement, repeated here because the tests import
# device.py directly - Home Assistant only installs manifest requirements when it loads
# the integration, which the test venv never does.
#
# This version and the one in custom_components/arozen_eon/manifest.json must be the same
# string, and tests/test_packaging.py fails if they are not. The manifest is what gets
# installed into a user's Home Assistant; this file is what CI proves the code works
# against. A bump moves both or neither - a manifest permitting a version the suite never
# ran is the failure this pin exists to prevent, and it is silent from here.
tinytuya==1.20.0

# Coverage reporting. Not a quality score - the job it was added for is narrower and
# structural: until it existed, a module could fall out of the suite entirely and nothing
# said so. __init__.py and device.py had done exactly that, and the symptom was a green
# run. The flags live in .github/workflows/ci.yml rather than in pyproject's `addopts`,
# so that a bare `pytest` still works in an environment without the plugin.
pytest-cov==7.1.0

# Pinned on purpose: an unpinned ruff picks up new rules on release and turns CI red
# on a morning nobody changed anything.
ruff==0.16.3
